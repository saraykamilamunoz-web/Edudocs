<script>
    function cargarTabla() {
        let solicitudes = JSON.parse(localStorage.getItem('solicitudes_db')) || [];
        let html = '';

        if(solicitudes.length === 0) {
            html = `<tr><td colspan="6" style="text-align:center; color:#888;">No hay solicitudes registradas en la bandeja.</td></tr>`;
        } else {
            solicitudes.forEach(s => {
                html += `<tr>
                    <td>${s.tipoDoc || 'T.I.'} ${s.doc}</td>
                    <td>${s.nombre}</td>
                    <td>${s.correo || 'N/A'}</td>
                    <td>${s.tramite}</td>
                    <td><span class="badge ${s.estado}">${s.estado}</span></td>
                    <td>
                        ${s.estado === 'Pendiente' 
                            ? `<button class="btn-aprobar" onclick="aprobar(${s.id})">Aprobar</button>` 
                            : `<button class="btn-ver" onclick="verCertificado(${s.id})">📄 Generar Documento</button>`}
                    </td>
                </tr>`;
            });
        }
        document.getElementById('tablaSolicitudes').innerHTML = html;
    }

    function aprobar(id) {
        let solicitudes = JSON.parse(localStorage.getItem('solicitudes_db')) || [];
        let item = solicitudes.find(s => s.id === id);
        if(item) {
            item.estado = 'Aprobado';
            localStorage.setItem('solicitudes_db', JSON.stringify(solicitudes));
            cargarTabla();
        }
    }

    function limpiarBandeja() {
        if(confirm('¿Estás seguro de que deseas eliminar todas las solicitudes de la bandeja?')) {
            localStorage.removeItem('solicitudes_db');
            cargarTabla();
        }
    }

    function verCertificado(id) {
        let solicitudes = JSON.parse(localStorage.getItem('solicitudes_db')) || [];
        let s = solicitudes.find(item => item.id === id);

        if (s) {
            const hoy = new Date();
            const meses = ["enero", "febrero", "marzo", "abril", "mayo", "junio", "julio", "agosto", "septiembre", "octubre", "noviembre", "diciembre"];
            
            if(s.tramite === "Paz y Salvo Académico") {
                document.getElementById('certCuerpo').innerHTML = `
                    Que, <b>${s.nombre.toUpperCase()}</b> identificado(a) con ${s.tipoDoc || 'T.I.'} No. <span>${s.doc}</span> cursó el grado <b>${s.curso || 'ONCE (11°)'}</b> de ${s.nivel || 'Educación Media Académica'} y se encuentra a la fecha <b>PAZ Y SALVO</b> por todo concepto académico y administrativo con la Institución Educativa.
                `;
                document.getElementById('certDetalles').style.display = 'none';
            } else {
                document.getElementById('certCuerpo').innerHTML = `
                    Que, <b>${s.nombre.toUpperCase()}</b> identificado(a) con ${s.tipoDoc || 'T.I.'} No. <span>${s.doc}</span> se encuentra matriculado(a) en esta Institución Educativa cursando el grado <b>${s.curso || 'ONCE (11°)'}</b> de ${s.nivel || 'Educación Media Académica'}.
                `;
                document.getElementById('certDetalles').style.display = 'block';
            }

            document.getElementById('certFecha').innerText = `La presente constancia se expide a solicitud del interesado(a), a los ${hoy.getDate()} días del mes de ${meses[hoy.getMonth()]} de ${hoy.getFullYear()} en el municipio de Sopó.`;

            document.getElementById('modalCertificado').style.display = 'block';
        }
    }

    function cerrarModal() {
        document.getElementById('modalCertificado').style.display = 'none';
    }

    cargarTabla();
    setInterval(cargarTabla, 2000);
</script>
